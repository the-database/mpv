cmake_minimum_required(VERSION 3.20)
project(ajn_mvtools LANGUAGES CXX)
add_library(ajn-mvtools SHARED detector.cpp
    vendor/CopyCode.cpp vendor/Fakery.cpp vendor/GroupOfPlanes.cpp
    vendor/Luma.cpp vendor/MVFrame.cpp vendor/PlaneOfBlocks.cpp vendor/SADFunctions.cpp)
target_compile_features(ajn-mvtools PRIVATE cxx_std_17)
target_compile_definitions(ajn-mvtools PRIVATE _CRT_SECURE_NO_WARNINGS AJN_MVTOOLS_BUILD)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    # SSE2 is part of the x64 baseline. Keep other architectures on the
    # original C kernels; do not enable the full plugin's assembly dispatch.
    target_compile_definitions(ajn-mvtools PRIVATE AJN_MVTOOLS_SSE2)
endif()
if(MSVC)
    set_property(TARGET ajn-mvtools PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(ajn-mvtools PRIVATE /O2 /EHsc /W3 /FIcstddef)
else()
    target_compile_options(ajn-mvtools PRIVATE -O3 -include cstddef)
endif()
install(TARGETS ajn-mvtools RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES LICENSE UPSTREAM.md DESTINATION licenses/mvtools)
enable_testing()
add_executable(mvtools-test test_detector.cpp vendor/SADFunctions.cpp vendor/Luma.cpp)
if(MSVC)
    target_compile_options(mvtools-test PRIVATE /FIcstddef)
else()
    target_compile_options(mvtools-test PRIVATE -include cstddef)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    target_compile_definitions(mvtools-test PRIVATE AJN_MVTOOLS_SSE2)
endif()
target_link_libraries(mvtools-test PRIVATE ajn-mvtools)
target_compile_features(mvtools-test PRIVATE cxx_std_17)
add_test(NAME mvtools-detector COMMAND mvtools-test)
